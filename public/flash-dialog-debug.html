🚨 弹窗闪现问题调试

专门用于调试弹窗一闪而过的问题

🔥 紧急调试：弹窗闪现问题

弹窗一闪而过通常是以下原因：

  • 自动关闭逻辑 - 弹窗被代码自动关闭
  • 事件冲突 - 点击事件同时触发了开启和关闭
  • 状态管理问题 - React状态快速变化
  • CSS动画问题 - 动画导致弹窗消失

步骤1：在主应用页面运行闪现调试代码

1. 打开 http://localhost:5173/adapt

2. 按 F12 打开控制台

3. 复制并运行以下调试代码：

// 🚨 弹窗闪现问题调试脚本 console.log('🔍 开始调试弹窗闪现问题...'); // 监控弹窗状态变化 let dialogObserver = null; let dialogStateLog = []; // 记录状态变化 function logDialogState(action, element) { const timestamp = Date.now(); const state = { time: new Date().toLocaleTimeString() + '.' + (timestamp % 1000), action: action, element: element ? { id: element.id, className: element.className, dataState: element.getAttribute('data-state'), display: window.getComputedStyle(element).display, visibility: window.getComputedStyle(element).visibility, opacity: window.getComputedStyle(element).opacity } : null }; dialogStateLog.push(state); console.log(`📊 [${state.time}] ${action}`, state.element); } // 监控DOM变化 function startDialogMonitoring() { // 监控弹窗元素的添加和移除 dialogObserver = new MutationObserver((mutations) => { mutations.forEach((mutation) => { // 监控添加的节点 mutation.addedNodes.forEach((node) => { if (node.nodeType === 1) { // Element node if (node.matches && node.matches('[role="dialog"]')) { logDialogState('弹窗元素添加', node); } // 检查子元素 const dialogs = node.querySelectorAll && node.querySelectorAll('[role="dialog"]'); if (dialogs) { dialogs.forEach(dialog => logDialogState('弹窗子元素添加', dialog)); } } }); // 监控移除的节点 mutation.removedNodes.forEach((node) => { if (node.nodeType === 1) { // Element node if (node.matches && node.matches('[role="dialog"]')) { logDialogState('弹窗元素移除', node); } } }); // 监控属性变化 if (mutation.type === 'attributes' && mutation.target.matches('[role="dialog"]')) { logDialogState(`属性变化: ${mutation.attributeName}`, mutation.target); } }); }); // 开始监控 dialogObserver.observe(document.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['data-state', 'class', 'style'] }); console.log('✅ 弹窗监控已启动'); } // 监控点击事件 function monitorClickEvents() { document.addEventListener('click', (e) => { const target = e.target; const button = target.closest('button'); if (button && button.textContent && button.textContent.includes('历史记录')) { console.log('🔍 历史记录按钮被点击', { button: button, timestamp: new Date().toLocaleTimeString() }); // 延迟检查弹窗状态 setTimeout(() => { const dialogs = document.querySelectorAll('[role="dialog"]'); console.log(`📊 点击后50ms: 找到 ${dialogs.length} 个弹窗`); dialogs.forEach((dialog, index) => { logDialogState(`点击后50ms检查-弹窗${index+1}`, dialog); }); }, 50); setTimeout(() => { const dialogs = document.querySelectorAll('[role="dialog"]'); console.log(`📊 点击后200ms: 找到 ${dialogs.length} 个弹窗`); dialogs.forEach((dialog, index) => { logDialogState(`点击后200ms检查-弹窗${index+1}`, dialog); }); }, 200); setTimeout(() => { const dialogs = document.querySelectorAll('[role="dialog"]'); console.log(`📊 点击后1000ms: 找到 ${dialogs.length} 个弹窗`); dialogs.forEach((dialog, index) => { logDialogState(`点击后1000ms检查-弹窗${index+1}`, dialog); }); }, 1000); } }); console.log('✅ 点击事件监控已启动'); } // 检查现有弹窗 function checkExistingDialogs() { const dialogs = document.querySelectorAll('[role="dialog"]'); console.log(`📊 当前页面弹窗数量: ${dialogs.length}`); dialogs.forEach((dialog, index) => { logDialogState(`现有弹窗${index+1}`, dialog); }); } // 查看状态日志 function showStateLog() { console.log('📊 弹窗状态变化日志:'); dialogStateLog.forEach((state, index) => { console.log(`${index + 1}. [${state.time}] ${state.action}`, state.element); }); if (dialogStateLog.length === 0) { console.log('📊 暂无状态变化记录'); } } // 停止监控 function stopMonitoring() { if (dialogObserver) { dialogObserver.disconnect(); console.log('🛑 弹窗监控已停止'); } } // 启动所有监控 startDialogMonitoring(); monitorClickEvents(); checkExistingDialogs(); // 提供手动函数 window.dialogFlashDebug = { showLog: showStateLog, stop: stopMonitoring, restart: () => { stopMonitoring(); dialogStateLog = []; startDialogMonitoring(); monitorClickEvents(); console.log('🔄 监控已重启'); } }; console.log('✅ 弹窗闪现调试已启动'); console.log('💡 现在点击历史记录按钮，然后运行 dialogFlashDebug.showLog() 查看详细日志'); console.log('💡 使用 dialogFlashDebug.stop() 停止监控'); console.log('💡 使用 dialogFlashDebug.restart() 重启监控');

步骤2：测试并分析结果

运行调试代码后：

  1. 点击历史记录按钮
  2. 观察控制台输出
  3. 运行 dialogFlashDebug.showLog() 查看详细日志
  4. 将完整的日志发送给我

步骤3：可能的问题和解决方案

如果日志显示：

  • "弹窗元素添加" → "弹窗元素移除" → 弹窗被代码自动关闭
  • "data-state: open" → "data-state: closed" → React状态快速变化
  • 只有添加没有移除，但弹窗消失 → CSS动画或样式问题
  • 多个快速的属性变化 → 可能有冲突的修复逻辑

🔍 弹窗位置检查工具

✅ 好消息：弹窗没有闪现，是正常显示的！

现在我们需要检查弹窗的实际位置和可见性。

请在主应用页面控制台运行以下代码：

// 🔍 检查弹窗位置和可见性 function checkDialogPosition() { const dialog = document.querySelector('[role="dialog"]'); if (!dialog) { console.log('❌ 未找到弹窗'); return; } const rect = dialog.getBoundingClientRect(); const viewport = { width: window.innerWidth, height: window.innerHeight }; const style = window.getComputedStyle(dialog); console.log('📊 弹窗详细信息:'); console.log('位置:', { top: rect.top, left: rect.left, right: rect.right, bottom: rect.bottom, width: rect.width, height: rect.height }); console.log('视窗:', viewport); console.log('样式:', { position: style.position, top: style.top, left: style.left, transform: style.transform, zIndex: style.zIndex, display: style.display, visibility: style.visibility, opacity: style.opacity }); // 检查是否在视窗内 const isInViewport = rect.top >= 0 && rect.left >= 0 && rect.bottom <= viewport.height && rect.right <= viewport.width; console.log('🎯 可见性分析:'); console.log('是否在视窗内:', isInViewport); if (!isInViewport) { console.log('⚠️ 弹窗位置问题:'); if (rect.top < 0) console.log('- 超出顶部边界'); if (rect.left < 0) console.log('- 超出左侧边界'); if (rect.bottom > viewport.height) console.log('- 超出底部边界'); if (rect.right > viewport.width) console.log('- 超出右侧边界'); } // 计算应该的中心位置 const expectedCenterX = viewport.width / 2; const expectedCenterY = viewport.height / 2; const actualCenterX = rect.left + rect.width / 2; const actualCenterY = rect.top + rect.height / 2; console.log('🎯 居中分析:'); console.log('期望中心:', { x: expectedCenterX, y: expectedCenterY }); console.log('实际中心:', { x: actualCenterX, y: actualCenterY }); console.log('偏移量:', { x: Math.abs(actualCenterX - expectedCenterX).toFixed(1) + 'px', y: Math.abs(actualCenterY - expectedCenterY).toFixed(1) + 'px' }); // 强制修复位置 console.log('🔧 尝试强制修复位置...'); dialog.style.setProperty('position', 'fixed', 'important'); dialog.style.setProperty('top', '50%', 'important'); dialog.style.setProperty('left', '50%', 'important'); dialog.style.setProperty('transform', 'translate(-50%, -50%)', 'important'); dialog.style.setProperty('z-index', '1000000', 'important'); dialog.style.setProperty('inset', 'unset', 'important'); // 验证修复效果 setTimeout(() => { const newRect = dialog.getBoundingClientRect(); const newCenterX = newRect.left + newRect.width / 2; const newCenterY = newRect.top + newRect.height / 2; console.log('✅ 修复后位置:', { top: newRect.top, left: newRect.left, centerX: newCenterX, centerY: newCenterY, 偏移量: { x: Math.abs(newCenterX - expectedCenterX).toFixed(1) + 'px', y: Math.abs(newCenterY - expectedCenterY).toFixed(1) + 'px' } }); const isNowCentered = Math.abs(newCenterX - expectedCenterX) < 10 && Math.abs(newCenterY - expectedCenterY) < 10; if (isNowCentered) { console.log('🎉 弹窗已成功居中！'); } else { console.log('❌ 弹窗仍未正确居中'); } }, 100); } // 运行检查 checkDialogPosition();

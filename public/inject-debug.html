🔍 主应用调试工具

这个工具将帮助您在主应用页面中调试历史记录弹窗问题

步骤1：打开主应用页面

在新标签页中打开：http://localhost:5173/adapt

步骤2：在主应用页面的控制台中运行调试代码

1. 在主应用页面按 F12 打开开发者工具

2. 切换到 Console 标签

3. 复制并粘贴以下代码，然后按回车：

// 🔍 历史记录弹窗调试脚本 console.log('🚀 开始调试历史记录弹窗...'); // 查找历史记录按钮 function findHistoryButton() { const buttons = Array.from(document.querySelectorAll('button')); const historyButton = buttons.find(btn => btn.textContent && btn.textContent.includes('历史记录') ); if (historyButton) { console.log('✅ 找到历史记录按钮'); console.log('按钮文本:', historyButton.textContent.trim()); console.log('按钮类名:', historyButton.className); return historyButton; } else { console.log('❌ 未找到历史记录按钮'); console.log('页面上的所有按钮文本:', buttons.map(btn => btn.textContent?.trim()).filter(Boolean)); return null; } } // 检查弹窗元素 function checkDialogElements() { console.log('🔍 检查弹窗元素...'); const selectors = [ '[role="dialog"]', '.enhanced-history-dialog', '[data-radix-dialog-content]', '[data-state="open"]', '[data-radix-portal]' ]; let foundAny = false; selectors.forEach(selector => { const elements = document.querySelectorAll(selector); if (elements.length > 0) { console.log(`✅ ${selector}: 找到 ${elements.length} 个元素`); foundAny = true; elements.forEach((el, index) => { const rect = el.getBoundingClientRect(); const style = window.getComputedStyle(el); console.log(` 元素 ${index + 1}:`, { className: el.className, id: el.id, 位置: `top=${rect.top.toFixed(1)}, left=${rect.left.toFixed(1)}`, 尺寸: `${rect.width.toFixed(1)}x${rect.height.toFixed(1)}`, 显示: style.display, 可见性: style.visibility, 透明度: style.opacity, 层级: style.zIndex }); }); } else { console.log(`❌ ${selector}: 找到 0 个元素`); } }); return foundAny; } // 测试按钮点击 function testButtonClick() { const button = findHistoryButton(); if (button) { console.log('🔍 准备点击历史记录按钮...'); // 添加点击监听器 button.addEventListener('click', function() { console.log('✅ 历史记录按钮被点击'); // 检查弹窗是否出现 setTimeout(() => { console.log('--- 点击后100ms检查 ---'); checkDialogElements(); }, 100); setTimeout(() => { console.log('--- 点击后500ms检查 ---'); checkDialogElements(); }, 500); }); // 点击按钮 button.click(); } } // 执行初始检查 console.log('=== 初始状态检查 ==='); findHistoryButton(); checkDialogElements(); console.log('=== 测试按钮点击 ==='); testButtonClick(); console.log('✅ 调试完成！请查看上面的输出结果'); console.log('💡 如果需要重新测试，运行: testButtonClick()');

步骤3：分析结果

根据控制台输出，我们可以确定：

  • 如果找到历史记录按钮但没有弹窗元素 → React组件渲染问题
  • 如果没有找到历史记录按钮 → 页面加载问题
  • 如果有JavaScript错误 → 代码错误问题
  • 如果找到弹窗但位置异常 → CSS定位问题
